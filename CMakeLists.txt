cmake_minimum_required(VERSION 3.16)
project(sorted_set LANGUAGES CXX)
add_library(sorted_set INTERFACE)
target_include_directories(sorted_set INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(sorted_set INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(containers_test tests/containers.cpp)
  target_link_libraries(containers_test PRIVATE sorted_set)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(containers_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME containers COMMAND containers_test)
endif()
add_executable(sorted_set_example examples/basic.cpp)
target_link_libraries(sorted_set_example PRIVATE sorted_set)
